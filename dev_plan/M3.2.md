## CC，请执行以下任务：M3.2 — 任务系统抽象重构

M3.1 已完成。本次任务将“任务”抽象为前后端统一原语：

- **后端**：`BaseTask` 类定义任务的生命周期、输入、执行、输出，所有任务类型继承并注册
- **前端**：`TaskCard` 组件独立管理自身状态、轮询、点击、重试、删除，成为任务的“前端化身”

---

## 一、核心设计原则

1. **任务是一等公民**：每个任务具有完整生命周期（pending → running → completed/failed）
2. **前后端统一契约**：任务通过 `task_id` 关联，前后端使用统一数据结构
3. **卡片独立**：每个任务卡片承载自己的状态、轮询、交互行为，不依赖父组件传递方法
4. **易扩展**：新增任务类型只需后端继承 `BaseTask` + 前端配置组件

```python
# backend/app/tasks/base.py
from abc import ABC, abstractmethod
from pydantic import BaseModel
from typing import Any, Optional
from enum import Enum
from datetime import datetime
import uuid

class TaskStatus(str, Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    CANCELLED = "cancelled"

class TaskInput(BaseModel):
    """任务输入参数基类。各任务类型继承并定义字段。"""
    pass

class TaskOutput(BaseModel):
    """任务输出结果基类。各任务类型继承并定义字段。"""
    content_type: str  # 决定前端渲染组件：'questions' | 'knowledge_graph' | 'document' | 'chat'

class BaseTask(ABC):
    task_type: str = "base"
    display_name: str = "任务"
    icon: str = "📋"
    description: str = ""
    timeout_seconds: int = 300  # 默认5分钟超时

    def __init__(self, task_id: str, document_id: str, input_params: dict):
        self.task_id = task_id
        self.document_id = document_id
        self.status = TaskStatus.PENDING
        self.input = self.validate_input(input_params)
        self.output: Optional[TaskOutput] = None
        self.error: Optional[str] = None
        self.progress = 0
        self.progress_text = ""
        self.created_at = datetime.now()
        self.updated_at = datetime.now()

    @abstractmethod
    def validate_input(self, params: dict) -> TaskInput:
        """校验输入参数，返回 Pydantic 模型"""
        pass

    @abstractmethod
    async def execute(self) -> TaskOutput:
        """执行任务核心逻辑，返回输出结果"""
        pass

    @abstractmethod
    def render(self, output: TaskOutput) -> dict:
        """将输出转为前端渲染数据：{"content_type": "...", "title": "...", "data": {...}}"""
        pass

    def to_card(self) -> dict:
        """转换为前端卡片数据（统一格式）"""
        return {
            "task_id": self.task_id,
            "task_type": self.task_type,
            "icon": self.icon,
            "title": self.display_name,
            "status": self.status.value,
            "progress": self.progress,
            "progress_text": self.progress_text,
            "created_at": self.created_at.isoformat(),
            "result": self.render(self.output) if self.output else None,
            "error": self.error
        }
```

```python
# backend/app/tasks/registry.py
_task_registry: dict[str, type[BaseTask]] = {}

def register_task(task_class: type[BaseTask]) -> type[BaseTask]:
    _task_registry[task_class.task_type] = task_class
    return task_class

def get_task(task_type: str, task_id: str, document_id: str, params: dict) -> BaseTask:
    if task_type not in _task_registry:
        raise ValueError(f"Unknown task type: {task_type}")
    return _task_registry[task_type](task_id, document_id, params)

def get_all_task_types() -> dict:
    return {
        t: {
            "task_type": cls.task_type,
            "display_name": cls.display_name,
            "icon": cls.icon,
            "description": cls.description
        }
        for t, cls in _task_registry.items()
    }
```

```python
from pydantic import BaseModel
from typing import List, Optional

class QuestionTaskInput(TaskInput):
    document_id: str
    entity_ids: List[str]
    scenario: str = "section_review"

        "choice": 0.30,
        "multi_choice": 0.20,
        "fill": 0.20,
        "short_answer": 0.15,
        "essay": 0.15
    }
    total_count: int = 18

class QuestionTaskOutput(TaskOutput):
    content_type: str = "questions"
    title: str
    questions: List[dict]
    metadata: dict

@register_task
class QuestionTask(BaseTask):
    task_type = "question_generation"
    display_name = "出题"
    icon = "📝"
    description = "基于选中的知识点生成训练题目"

    def validate_input(self, params: dict) -> QuestionTaskInput:

            title=f"{self.input.scenario}训练题",
            questions=questions,
            metadata={
                "total": len(questions),
                "types": self.input.types
            }
        )

    def render(self, output: QuestionTaskOutput) -> dict:
        return {
            "content_type": "questions",
            "title": output.title,
            "data": {
                "questions": output.questions,
                "metadata": output.metadata
```

### 2.5 任务管理器

```python
# backend/app/core/task_manager.py
from app.tasks.registry import get_task, get_all_task_types
from app.tasks.base import BaseTask, TaskStatus
import asyncio
import uuid

class TaskManager:
    def __init__(self):
        self._tasks: dict[str, BaseTask] = {}
        self._persist = ...  # 数据库持久化适配器

    async def create_task(self, task_type: str, document_id: str, params: dict) -> dict:
        task_id = str(uuid.uuid4())
        task = get_task(task_type, task_id, document_id, params)
        self._tasks[task_id] = task
        await self._persist.save(task)
        asyncio.create_task(self._run_task(task_id))
        return task.to_card()

    async def _run_task(self, task_id: str):
        task = self._tasks[task_id]
        task.status = TaskStatus.RUNNING
        await self._persist.update(task)

        try:
            # 带超时控制
            output = await asyncio.wait_for(task.execute(), timeout=task.timeout_seconds)
            task.output = output
            task.status = TaskStatus.COMPLETED
            task.progress = 100
            task.progress_text = "已完成"
        except asyncio.TimeoutError:
            task.status = TaskStatus.FAILED
            task.error = "任务执行超时"
            task.progress_text = "超时"
        except Exception as e:
            task.status = TaskStatus.FAILED
            task.error = str(e)
            task.progress_text = f"失败: {str(e)[:50]}"
        finally:
            await self._persist.update(task)

    def get_task_status(self, task_id: str) -> dict:
        task = self._tasks.get(task_id)
        if not task:
            return {"status": "not_found"}
        result = task.render(task.output) if task.output else None
        return {
            "task_id": task.task_id,
            "status": task.status.value,
            "progress": task.progress,
            "progress_text": task.progress_text,
            "result": result,
            "error": task.error
        }

    def get_all_tasks(self, document_id: str = None) -> list[dict]:
        tasks = list(self._tasks.values())
        if document_id:
            tasks = [t for t in tasks if t.document_id == document_id]
        tasks.sort(key=lambda t: t.created_at, reverse=True)
        return [t.to_card() for t in tasks]

    def get_task_types(self) -> dict:
        return get_all_task_types()

    async def retry_task(self, task_id: str) -> dict:
        """重置并重新执行任务"""
        task = self._tasks.get(task_id)
        if not task:
            raise ValueError("Task not found")
        task.status = TaskStatus.PENDING
        task.progress = 0
        task.error = None
        await self._persist.update(task)
        asyncio.create_task(self._run_task(task_id))
        return task.to_card()

    async def delete_task(self, task_id: str) -> bool:
        """删除任务卡片（软删除）"""
        if task_id in self._tasks:
            del self._tasks[task_id]
            await self._persist.delete(task_id)
            return True
        return False
```

```python
from fastapi import APIRouter, Query
```

## 三、前端：独立任务卡片

### 3.1 设计原则

- 每个卡片持有 `task_id`
- 卡片在挂载时加载自身状态，自动启动轮询（如果状态为 pending/running）
- 卡片根据状态自行渲染（进度、颜色、交互）
- 点击、重试、删除全部由卡片自身处理，不依赖父组件传递方法

### 3.2 TaskCard 组件（核心）

```vue
<!-- frontend/src/components/RightPanel/TaskCard.vue -->
<template>
  <div 
    class="task-card"
    :class="{
      'is-pending': status === 'pending',
      'is-running': status === 'running',
      'is-completed': status === 'completed',
      'is-failed': status === 'failed',
      'is-clickable': status === 'completed' || status === 'failed'
    }"
    @click="handleClick"
  >
    <div class="card-icon">{{ icon }}</div>
    <div class="card-body">
      <div class="card-title">{{ title }}</div>
      <div class="card-meta">
        <span class="status-badge" :class="`badge-${status}`">
          {{ statusLabel }}
        </span>
        <span class="card-time">{{ formatTime(createdAt) }}</span>
      </div>
      <div v-if="status === 'running'" class="progress-wrapper">
        <div class="progress-bar" :style="{ width: progress + '%' }"></div>
        <span class="progress-text">{{ progressText }}</span>
      </div>
      <div v-if="status === 'failed'" class="error-text">
        {{ error || '任务失败，点击重试' }}
      </div>
    </div>
    <div class="card-actions">
      <button 
        v-if="status === 'failed'" 
        class="retry-btn" 
        @click.stop="handleRetry"
        title="重试任务"
      >
        ↻
      </button>
      <button 
        class="delete-btn" 
        @click.stop="handleDelete"
        title="删除卡片"
      >
        ✕
      </button>
    </div>
  </div>
</template>

<script setup>
import { ref, computed, onMounted, onUnmounted } from 'vue'
import { useTaskStore } from '@/stores/useTaskStore'
import { useContentStore } from '@/stores/useContentStore'

const props = defineProps({
  taskId: { type: String, required: true },
  // 初始数据（由父组件传入，用于 SSR 或列表预加载）
  initialStatus: { type: Object, default: null }
})

const emit = defineEmits(['deleted'])

const taskStore = useTaskStore()
const contentStore = useContentStore()

// 卡片自身状态
const status = ref('pending')
const title = ref('任务')
const icon = ref('📋')
const progress = ref(0)
const progressText = ref('')
const error = ref(null)
const createdAt = ref(null)
const result = ref(null)

let pollTimer = null

// 状态标签
const statusLabel = computed(() => {
  const map = {
    pending: '⏳ 排队中',
    running: '🔄 运行中',
    completed: '✅ 已完成',
    failed: '❌ 失败'
  }
  return map[status.value] || status.value
})

// 加载状态
const loadStatus = async () => {
  try {
    const data = await taskStore.fetchTaskStatus(props.taskId)
    status.value = data.status
    title.value = data.title || '任务'
    icon.value = data.icon || '📋'
    progress.value = data.progress || 0
    progressText.value = data.progress_text || ''
    error.value = data.error || null
    createdAt.value = data.created_at || null
    result.value = data.result || null
    return data
  } catch (e) {
    console.error('Failed to load task status:', e)
  }
}

// 轮询控制
const startPolling = () => {
  if (pollTimer) return
  pollTimer = setInterval(async () => {
    await loadStatus()
    if (status.value === 'completed' || status.value === 'failed') {
      stopPolling()
    }
  }, 3000)
}

const stopPolling = () => {
  if (pollTimer) {
    clearInterval(pollTimer)
    pollTimer = null
  }
}

// 交互处理
const handleClick = () => {
  if (status.value === 'completed') {
    // 加载结果到中栏
    contentStore.loadContent(result.value)
  } else if (status.value === 'failed') {
    handleRetry()
  }
  // pending/running 不可点击
}

const handleRetry = async () => {
  status.value = 'pending'
  progress.value = 0
  progressText.value = '重试中...'
  await taskStore.retryTask(props.taskId)
  startPolling()
}

const handleDelete = async () => {
  if (confirm('确定要删除此任务卡片吗？')) {
    await taskStore.deleteTask(props.taskId)
    emit('deleted')
  }
}

// 生命周期
onMounted(async () => {
  // 如果父组件传入了初始状态，直接使用（减少 API 调用）
  if (props.initialStatus) {
    status.value = props.initialStatus.status
    title.value = props.initialStatus.title || '任务'
    icon.value = props.initialStatus.icon || '📋'
    progress.value = props.initialStatus.progress || 0
    progressText.value = props.initialStatus.progress_text || ''
    error.value = props.initialStatus.error || null
    createdAt.value = props.initialStatus.created_at || null
    result.value = props.initialStatus.result || null
  } else {
    await loadStatus()
  }

  if (status.value === 'pending' || status.value === 'running') {
    startPolling()
  }
})

onUnmounted(() => {
  stopPolling()
})
</script>
```
