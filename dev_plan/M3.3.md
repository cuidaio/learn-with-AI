M3.2 已完成，任务抽象已就位。本次任务聚焦于降低任务与基础设施之间的**间接耦合**，将任务执行层对具体依赖的直接导入改为通过接口注入。

## 一、背景与目标

### 1.1 当前耦合点

M3.2 中任务类直接导入具体实现：

```python
# question_task.py
from app.core.question_generator import generate_questions  # 直接依赖具体函数

# graph_task.py  
from app.core.entity_extractor import extract_entities      # 直接依赖
from app.core.relation_extractor import extract_relations   # 直接依赖

# task_manager.py
self._persist = ...  # 直接依赖数据库实现
```

**问题**：
- 测试困难（需要 mock 具体函数）
- 更换实现（如换 LLM 提供商）需要修改任务代码
- 任务类承担了“依赖管理”的职责，违反了单一职责

### 1.2 目标

1. **任务类只关心执行逻辑**，不关心依赖从哪里来
2. **外部依赖通过构造函数注入**，方便替换和测试
3. **所有基础设施（LLM、检索、数据库、出题、图谱）抽象为接口**


## 二、核心设计

### 2.1 依赖注入架构

```
┌─────────────────────────────────────────────────────────────┐
│  BaseTask                                                   │
│  ├── 输入参数 + 输出参数                                    │
│  ├── execute() 抽象方法                                    │
│  └── 所有外部依赖 → 通过构造函数注入（非import）          │
└─────────────────────────────────────────────────────────────┘
                              ▲
                              │
┌─────────────────────────────────────────────────────────────┐
│  QuestionTask                                               │
│  ├── __init__(question_generator: QuestionGenerator)       │
│  └── execute() → self._question_generator.generate(...)    │
└─────────────────────────────────────────────────────────────┘
                              ▲
                              │
┌─────────────────────────────────────────────────────────────┐
│  TaskManager                                                │
│  ├── __init__(task_store: TaskStore, executor: Executor)   │
│  └── create_task() → 调用工厂 + 注入依赖                   │
└─────────────────────────────────────────────────────────────┘
```

### 2.2 需要抽象的服务接口

| 服务 | 当前具体依赖 | 抽象接口名 |
|------|-------------|-----------|
| 任务持久化 | 直接操作数据库 | `TaskStore` |
| 出题生成 | `generate_questions()` | `QuestionGenerator` |
| 实体提取 | `extract_entities()` | `EntityExtractor` |
| 关系提取 | `extract_relations()` | `RelationExtractor` |
| LLM 调用 | `openai.ChatCompletion` | `LLMClient` |
| Embedding | `openai.Embedding` | `EmbeddingClient` |

## 四、接口定义详细设计

### 4.1 接口文件

```python
# backend/app/core/interfaces.py
from abc import ABC, abstractmethod
from typing import List, Dict, Any, Optional
import asyncio

# ─── 任务存储接口 ───
class TaskStore(ABC):
    @abstractmethod
    async def save(self, task: "BaseTask") -> None:
        pass
    
    @abstractmethod
    async def update(self, task: "BaseTask") -> None:
        pass
    
    @abstractmethod
    async def get(self, task_id: str) -> Optional["BaseTask"]:
        pass
    
    @abstractmethod
    async def list(self, document_id: Optional[str] = None) -> List["BaseTask"]:
        pass
    
    @abstractmethod
    async def delete(self, task_id: str) -> bool:
        pass

# ─── 出题生成器接口 ───
class QuestionGenerator(ABC):
    @abstractmethod
    async def generate(
        self,
        document_id: str,
        entity_ids: List[str],
        config: Dict[str, Any]
    ) -> List[Dict[str, Any]]:
        pass

# ─── 实体提取器接口 ───
class EntityExtractor(ABC):
    @abstractmethod
    async def extract(self, document_id: str) -> List[Dict[str, Any]]:
        pass

# ─── 关系提取器接口 ───
class RelationExtractor(ABC):
    @abstractmethod
    async def extract(self, document_id: str, entities: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
        pass

# ─── LLM 客户端接口 ───
class LLMClient(ABC):
    @abstractmethod
    async def chat_completion(
        self,
        messages: List[Dict[str, str]],
        temperature: float = 0.2,
        stream: bool = False
    ) -> Dict[str, Any]:
        pass
    
    @abstractmethod
    async def chat_completion_stream(
        self,
        messages: List[Dict[str, str]],
        temperature: float = 0.2
    ):
        pass

# ─── Embedding 客户端接口 ───
class EmbeddingClient(ABC):
    @abstractmethod
    async def embed(self, text: str) -> List[float]:
        pass
    
    @abstractmethod
    async def embed_batch(self, texts: List[str]) -> List[List[float]]:
        pass

# ─── 任务工厂接口 ───
class TaskFactory(ABC):
    @abstractmethod
    def create(
        self,
        task_type: str,
        task_id: str,
        document_id: str,
        params: dict
    ) -> "BaseTask":
        pass
```

### 4.3 任务管理器修改

```python
# backend/app/core/task_manager.py
from app.core.interfaces import TaskStore, TaskFactory

class TaskManager:
    def __init__(
        self,
        task_store: TaskStore,     # ← 注入，而非硬编码
        task_factory: TaskFactory, # ← 注入
    ):
        self._store = task_store
        self._factory = task_factory
        self._running_tasks: dict[str, BaseTask] = {}
    
    async def create_task(self, task_type: str, document_id: str, params: dict) -> dict:
        task_id = str(uuid.uuid4())
        # 通过工厂创建任务，工厂内部会注入所有依赖
        task = self._factory.create(task_type, task_id, document_id, params)
        self._running_tasks[task_id] = task
        await self._store.save(task)
        asyncio.create_task(self._run_task(task_id))
        return task.to_card()
```

# backend/app/core/di_container.py
from app.core.interfaces import *
from app.core.adapters import *
from app.core.config import settings

class DIContainer:
    def __init__(self):
        self._singletons = {}
    
    def get_task_store(self) -> TaskStore:
        if "task_store" not in self._singletons:
            self._singletons["task_store"] = PostgresTaskStore()
        return self._singletons["task_store"]
    
    def get_llm_client(self) -> LLMClient:
        if "llm_client" not in self._singletons:
            self._singletons["llm_client"] = OpenAILLMClient(
                base_url=settings.llm_base_url,
                api_key=settings.llm_api_key,
                model=settings.llm_model
            )
        return self._singletons["llm_client"]
    
    def get_embedding_client(self) -> EmbeddingClient:
        if "embedding_client" not in self._singletons:
            self._singletons["embedding_client"] = OpenAIEmbeddingClient(
                base_url=settings.embedding_base_url,
                api_key=settings.embedding_api_key,
                model=settings.embedding_model
            )
        return self._singletons["embedding_client"]
    
    def get_question_generator(self) -> QuestionGenerator:
        if "question_generator" not in self._singletons:
            self._singletons["question_generator"] = QuestionGeneratorImpl(
                llm_client=self.get_llm_client(),
                embedding_client=self.get_embedding_client()
            )
        return self._singletons["question_generator"]
    
    def get_entity_extractor(self) -> EntityExtractor:

            self._singletons["entity_extractor"] = EntityExtractorImpl(
                llm_client=self.get_llm_client()

### 4.5 工厂实现

```python
# backend/app/tasks/registry.py
from app.core.interfaces import TaskFactory
from app.tasks.question_task import QuestionTask
from app.tasks.graph_task import GraphTask

class RegistryBasedTaskFactory(TaskFactory):
    def __init__(
        self,
        question_generator,
        entity_extractor,
        relation_extractor,
        llm_client
    ):
        self._question_generator = question_generator
        self._entity_extractor = entity_extractor
        self._relation_extractor = relation_extractor
        self._llm_client = llm_client
    
    def create(self, task_type: str, task_id: str, document_id: str, params: dict) -> BaseTask:
        if task_type == "question_generation":
            return QuestionTask(
                task_id=task_id,
                document_id=document_id,
                input_params=params,
                question_generator=self._question_generator,
                llm_client=self._llm_client
            )
        elif task_type == "graph_generation":
            return GraphTask(
                task_id=task_id,
                document_id=document_id,
                input_params=params,
                entity_extractor=self._entity_extractor,
                relation_extractor=self._relation_extractor
            )
        else:
            raise ValueError(f"Unknown task type: {task_type}")
```
